# starlette is fastapi's dependency, pinned here so the advisories against

starlette==1.6.0

zarr==2.18.2
# numcodecs is zarr's dependency, not ours, but it has to be pinned here: zarr
# 2.18.2 asks only for numcodecs>=0.10, and numcodecs 0.16 dropped the blosc
# symbol (cbuffer_sizes) that zarr 2.x imports at module scope. Leave it
# unpinned and a fresh install resolves the newest numcodecs, after which
# `import zarr` raises ImportError and the whole app fails to start. 0.12.1 is
# the version the working Windows environment has.
numcodecs==0.12.1
numpy==2.0.2
pillow==12.3.0
python-multipart==0.0.32
# h5py writes the AnnData (.h5ad) export directly against the documented
# on-disk schema -- shipping the anndata package itself would drag pandas and
# scipy (~200 MB) into the bundle for one file format.
h5py==3.12.1
# ruamel, NOT PyYAML: annotation.notes.yaml and metadata.yml are hand-maintained
# by several people, and only a round-tripper preserves their comments, key
# order, quoting and ---/... markers. PyYAML would reformat every file it wrote.
ruamel.yaml==0.18.6
# Region prediction: the model runs through ONNX Runtime so the app needs no PyTorch.
# scipy.ndimage is the prediction rules' arithmetic (the same functions the validated
# batch used), and 1.14.1 is the release built against numpy 2.0.
# 1.19.2, not newer: its Mac wheel runs on macOS 11 (1.20+ need 13). On the idx=22
# model it gives the same labels as 1.23.2 on 20 of 20 sections (logits within 3e-6).
# It has no Python 3.13 wheel: builds stay on 3.11.
onnxruntime==1.19.2
scipy==1.14.1
